```java Java
    .builder()
    .apiHost("https://cdn.growthbook.io")
    .clientKey("<environment_key>") // replace with your client key

    public void onError(Throwable throwable) {
        System.out.println("Features refresh failed: " + throwable.getMessage());
```

```kotlin Kotlin
val featuresRepository = GBFeaturesRepository
    .builder()
    .apiHost("https://cdn.growthbook.io")
    .clientKey("<environment_key>") // replace with your client key
    .decryptionKey("<client-key-for-decrypting>") // optional, nullable
    .swrTtlSeconds(30)
    .build()
    .apply {
        // Optional callback for getting updates when features are refreshed
        onFeaturesRefresh(object : FeatureRefreshCallback {
            override fun onRefresh(featuresJson: String) {
                println("Features have been refreshed \n $featuresJson")
            }

            override fun onError(throwable: Throwable) {
                println("Features refresh failed: ${throwable.message}")
            }
        })
    }

// Fetch the features
try {
    featuresRepository.initialize()
} catch (e: FeatureFetchException) {
    // TODO: handle the exception
    e.printStackTrace()
}

// Initialize the GrowthBook SDK with the GBContext and features
val context = GBContext
    .builder()
    .featuresJson(featuresRepository.featuresJson)
    .attributesJson(userAttributes)
    .build()

val growthBook = GrowthBook(context)
```
